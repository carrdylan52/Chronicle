include(FetchContent)
# Khronos OpenXR SDK release-1.1.63, peeled release commit. Bundled JSON and generated sources.
set(BUILD_API_LAYERS OFF)
set(BUILD_TESTS OFF)
set(BUILD_CONFORMANCE_TESTS OFF)
set(BUILD_SDK_TESTS OFF)
set(BUILD_WITH_SYSTEM_JSONCPP OFF)
set(DYNAMIC_LOADER OFF)
FetchContent_Declare(dc_openxr
    GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK.git
    GIT_TAG f2448a8797c85814aa892efc1ab8707900fbcc78)
FetchContent_MakeAvailable(dc_openxr)

add_library(dcvr_runtime STATIC runtime.cpp frame_loop.cpp ${CMAKE_SOURCE_DIR}/port/src/gfx/requirements.cpp)
target_include_directories(dcvr_runtime PUBLIC ${CMAKE_SOURCE_DIR}/port/src)
target_link_libraries(dcvr_runtime PUBLIC openxr_loader Vulkan::Vulkan SDL3::SDL3 nlohmann_json::nlohmann_json)
target_compile_definitions(dcvr_runtime PUBLIC XR_USE_GRAPHICS_API_VULKAN)
target_compile_options(dcvr_runtime PRIVATE -Wall)
if(WIN32)
    target_link_libraries(dcvr_runtime PUBLIC advapi32 ole32)
endif()

add_executable(dcvr_probe main.cpp)
set_target_properties(dcvr_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
target_link_libraries(dcvr_probe PRIVATE dcvr_runtime)
target_compile_options(dcvr_probe PRIVATE -Wall)
add_executable(dcvr_game ${CMAKE_SOURCE_DIR}/port/src/main.cpp game_bridge.cpp room.cpp)
set_target_properties(dcvr_game PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
dc_link_game(dcvr_game)
target_include_directories(dcvr_game PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(dcvr_game PRIVATE DCVR_GAME)
target_link_libraries(dcvr_game PRIVATE dcvr_runtime)
add_dependencies(dcvr_game dc_lang dc_glyphs)
if(WIN32)
    target_link_options(dcvr_game PRIVATE -municode)
    target_sources(dcvr_game PRIVATE ${CMAKE_SOURCE_DIR}/port/icon/chronicle.rc)
endif()
add_executable(dcvr_room room_main.cpp room.cpp)
set_target_properties(dcvr_room PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
target_sources(dcvr_room PRIVATE $<TARGET_OBJECTS:dc_host>)
target_link_libraries(dcvr_room PRIVATE dcvr_runtime)
target_link_options(dcvr_room PRIVATE ${DC_LINK_OPTIONS})
target_compile_options(dcvr_room PRIVATE -Wall)
if(WIN32)
    target_link_options(dcvr_room PRIVATE -municode)
endif()

if(DC_BUILD_TESTS)
    add_executable(dcvr_tests frame_loop_test.cpp room.cpp)
    set_target_properties(dcvr_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
    target_sources(dcvr_tests PRIVATE $<TARGET_OBJECTS:dc_host>)
    target_link_libraries(dcvr_tests PRIVATE dcvr_runtime GTest::gtest_main)
    target_link_options(dcvr_tests PRIVATE ${DC_LINK_OPTIONS})
    target_compile_options(dcvr_tests PRIVATE -Wall)
    gtest_discover_tests(dcvr_tests DISCOVERY_MODE PRE_TEST PROPERTIES TIMEOUT 10)
endif()
